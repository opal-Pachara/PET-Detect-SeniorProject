def initialize_hardware(self):
    """Initialize all hardware components"""
    try:
        # Initialize Camera
        self.camera = cv2.VideoCapture(0)
        self.camera.set(cv2.CAP_PROP_FRAME_WIDTH, 1280)
        self.camera.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)
        
        # Initialize Stepper Motor
        self.stepper = StepperMotorController(
            step_pin=18, dir_pin=16, enable_pin=22
        )
        
        # Initialize LCD Display
        if LCD_AVAILABLE:
            self.lcd = CharLCD(i2c_expander='PCF8574', address=0x27, port=1,
                              cols=16, rows=2, dotsize=8)
            self.lcd.clear()
            self.lcd.write_string("System Ready")
    
    except Exception as e:
        logger.error(f"Hardware initialization failed: {e}")

การทำงาน:
- ตั้งค่า USB Camera ด้วย OpenCV (ความละเอียด 1280x720)
- ตั้งค่า Stepper Motor (GPIO pins: step=18, dir=16, enable=22)
- ตั้งค่า LCD Display (I2C address 0x27, 16x2 characters)
- แสดงข้อความ "System Ready" เมื่อพร้อมใช้งาน

def create_rfid_helper(self):
    """สร้าง RFID helper script เพื่อหลีกเลี่ยง GPIO conflict"""
    script_content = '''#!/usr/bin/env python3
import time
import sys
import RPi.GPIO as GPIO
from mfrc522 import SimpleMFRC522

reader = SimpleMFRC522()

try:
    while True:
        id, text = reader.read()
        if id:
            print(f"RFID:{id}")
            sys.stdout.flush()
            time.sleep(1)
except KeyboardInterrupt:
    GPIO.cleanup()
'''
    
    with open('rfid_helper.py', 'w') as f:
        f.write(script_content)
    os.chmod('rfid_helper.py', 0o755)

การทำงาน:
- สร้างไฟล์ Python script สำหรับอ่าน RFID แยกต่างหาก
- ใช้ subprocess เพื่อหลีกเลี่ยง GPIO conflict กับ Stepper Motor
- ส่งข้อมูล RFID ผ่าน stdout pipe
- รันในลูปไม่สิ้นสุดเพื่อรอการแตะบัตร

def run(self):
    """Main loop"""
    logger.info("Starting PET Detect System...")
    
    try:
        while self.running:
            # เริ่ม RFID subprocess
            rfid_process = subprocess.Popen(
                ['python3', 'rfid_helper.py'],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True
            )
            
            # รอข้อมูล RFID
            for line in iter(rfid_process.stdout.readline, ''):
                if line.startswith('RFID:'):
                    rfid_id = line.strip().split(':')[1]
                    logger.info(f"RFID detected: {rfid_id}")
                    
                    # ประมวลผลการสแกน
                    self.process_scan_result(rfid_id)
                    
                    # หยุด subprocess
                    rfid_process.terminate()
                    break
                    
    except KeyboardInterrupt:
        logger.info("System shutdown requested")
    finally:
        self.cleanup()

การทำงาน:
- รันระบบในลูปไม่สิ้นสุด
- เริ่ม RFID subprocess เพื่อรอการแตะบัตร
- เมื่อได้ข้อมูล RFID จะเรียกฟังก์ชัน process_scan_result()
- หยุด subprocess หลังจากประมวลผลเสร็จ

def process_scan_result(self, rfid_id):
    """ประมวลผลการสแกน RFID และจับภาพ"""
    try:
        # แสดงสถานะบน LCD
        self.lcd_show_processing()
        
        # จับภาพจากกล้อง
        ret, frame = self.camera.read()
        if not ret:
            logger.error("Failed to capture image")
            return
        
        # บันทึกภาพ
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        image_filename = f"scan_{timestamp}_{rfid_id}.jpg"
        cv2.imwrite(image_filename, frame)
        
        # ส่งไป AI API
        result = self.send_to_ai_api(image_filename)
        if result:
            # คำนวณคะแนน
            self.calculate_and_display_score(result, rfid_id)
            
            # บันทึกลงฐานข้อมูล
            self.save_score_to_web(result, rfid_id)
            
            # ควบคุมมอเตอร์
            self.control_stepper(result)
        
    except Exception as e:
        logger.error(f"Error processing scan: {e}")

การทำงาน:
- แสดงข้อความ "Processing..." บน LCD
- จับภาพจาก USB Camera
- บันทึกภาพพร้อม timestamp และ RFID ID
- ส่งภาพไป AI API เพื่อวิเคราะห์
- คำนวณและแสดงคะแนน
- บันทึกผลลัพธ์ลงฐานข้อมูล
- ควบคุม Stepper Motor ตามผลลัพธ์

def send_to_ai_api(self, image_filename):
    """ส่งภาพไป AI API เพื่อวิเคราะห์"""
    try:
        url = f"{self.api_url}/api/scan"
        
        with open(image_filename, 'rb') as image_file:
            files = {'image': image_file}
            response = self.session.post(url, files=files, timeout=10)
        
        if response.status_code == 200:
            return response.json()
        else:
            logger.error(f"AI API error: {response.status_code}")
            return None
            
    except Exception as e:
        logger.error(f"Error sending to AI API: {e}")
        return None

การทำงาน:
- ส่งไฟล์ภาพไป AI API ผ่าน HTTP POST
- ใช้ multipart/form-data สำหรับส่งไฟล์
- รอ response จาก AI API (timeout 10 วินาที)
- Return JSON response หากสำเร็จ

def calculate_and_display_score(self, result, rfid_id):
    """คำนวณและแสดงคะแนนบน LCD"""
    if not result.get('success'):
        return
    
    # ดึงข้อมูลการตรวจจับ
    bottles = result.get('bottle_count', 0)
    cans = result.get('can_count', 0)
    caps = result.get('cap_count', 0)
    labels = result.get('label_count', 0)
    
    # คำนวณคะแนนตามระบบใหม่
    score = (bottles * 50) + (cans * 100) + (caps * (-10)) + (labels * (-10))
    
    # แสดงผลบน LCD
    self.lcd_show_results(bottles, cans, caps, labels)
    time.sleep(2)
    self.lcd_show_score(rfid_id, score)
    
    # แสดงผลบน Terminal
    print(f"\n=== ผลการวิเคราะห์ ===")
    print(f"RFID: {rfid_id}")
    print(f"ขวด PET: {bottles} อัน (+{bottles * 50} คะแนน)")
    print(f"กระป๋อง: {cans} อัน (+{cans * 100} คะแนน)")
    print(f"ฝา: {caps} อัน ({caps * (-10)} คะแนน)")
    print(f"ฉลาก: {labels} อัน ({labels * (-10)} คะแนน)")
    print(f"คะแนนรวม: {score} คะแนน")

การทำงาน:
- ดึงข้อมูลการตรวจจับจาก AI API response
- คำนวณคะแนนตามระบบใหม่:
  * ขวด PET: +50 คะแนน
  * กระป๋อง: +100 คะแนน
  * ฝา: -10 คะแนน
  * ฉลาก: -10 คะแนน
- แสดงผลบน LCD และ Terminal

def control_stepper(self, result_data):
    """ควบคุม Stepper Motor ตามผลการตรวจจับ"""
    if not self.stepper:
        return
    
    bottle_count = result_data.get('bottle_count', 0)
    can_count = result_data.get('can_count', 0)
    
    try:
        if bottle_count > 0 and can_count > 0:
            # พบทั้งขวดและกระป๋อง → หมุนขวา
            self.stepper.move_degrees(90, speed=150)
            time.sleep(2)
            
        elif bottle_count > 0:
            # พบขวดอย่างเดียว → หมุนซ้าย
            self.stepper.move_degrees(-90, speed=150)
            time.sleep(2)
            
        elif can_count > 0:
            # พบกระป๋องอย่างเดียว → หมุนขวา
            self.stepper.move_degrees(90, speed=150)
            time.sleep(2)
        
        # กลับตำแหน่งเดิม
        self.stepper.return_to_home(speed=200)
        time.sleep(1)
        
    except Exception as e:
        logger.error(f"Stepper control error: {e}")

การทำงาน:
- ตรวจสอบจำนวนขวดและกระป๋องที่ตรวจจับได้
- หมุนมอเตอร์ตามกฎ:
  * ขวด + กระป๋อง → หมุนขวา 90°
  * ขวดอย่างเดียว → หมุนซ้าย -90°
  * กระป๋องอย่างเดียว → หมุนขวา 90°
- กลับตำแหน่งเดิมหลังจากหมุนเสร็จ

def login_required(f):
    """Decorator สำหรับตรวจสอบการล็อกอิน"""
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if not session.get('admin_logged_in'):
            return redirect(url_for('admin_login'))
        return f(*args, **kwargs)
    return decorated_function

@app.route('/admin/login', methods=['GET', 'POST'])
def admin_login():
    """Admin login page"""
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        
        if not username or not password:
            return render_template('admin_login.html', error="กรุณากรอกชื่อผู้ใช้และรหัสผ่าน")
        
        # ตรวจสอบข้อมูลในฐานข้อมูล
        connection = get_database_connection()
        cursor = connection.cursor()
        
        if isinstance(connection, psycopg2.extensions.connection):
            cursor.execute("SELECT id, username, password_hash FROM admin_users WHERE username = %s", (username,))
        else:
            cursor.execute("SELECT id, username, password_hash FROM admin_users WHERE username = ?", (username,))
        
        user = cursor.fetchone()
        connection.close()
        
        if user:
            # ตรวจสอบรหัสผ่าน
            password_hash = hashlib.sha256(password.encode()).hexdigest()
            if password_hash == user['password_hash']:
                # เข้าสู่ระบบสำเร็จ
                session['admin_logged_in'] = True
                session['admin_username'] = user['username']
                session['admin_id'] = user['id']
                return redirect(url_for('admin_dashboard'))
        
        return render_template('admin_login.html', error="ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง")
    
    return render_template('admin_login.html', error=None)

การทำงาน:

class StepperMotorController:
    def __init__(self, step_pin=18, dir_pin=16, enable_pin=22):
        """Initialize stepper motor controller"""
        self.step_pin = step_pin
        self.dir_pin = dir_pin
        self.enable_pin = enable_pin
        self.current_position = 0  # ตำแหน่งปัจจุบัน (องศา)
        self.steps_per_revolution = 200  # จำนวน steps ต่อ 1 รอบ
        self.microsteps = 16  # Microstepping
        
        # ตั้งค่า GPIO
        GPIO.setmode(GPIO.BOARD)
        GPIO.setup(self.step_pin, GPIO.OUT)
        GPIO.setup(self.dir_pin, GPIO.OUT)
        GPIO.setup(self.enable_pin, GPIO.OUT)
        
        # เปิดใช้งานมอเตอร์
        GPIO.output(self.enable_pin, GPIO.LOW)  # LOW = Enable
        
        print(f"Stepper motor initialized on pins: STEP={self.step_pin}, DIR={self.dir_pin}, ENABLE={self.enable_pin}")

การทำงาน:
- ตั้งค่า GPIO pins สำหรับควบคุม Stepper Motor
- กำหนดพารามิเตอร์มอเตอร์ (steps per revolution, microsteps)
- เปิดใช้งานมอเตอร์โดยการ set enable pin เป็น LOW
- บันทึกตำแหน่งปัจจุบันของมอเตอร์

def move_degrees(self, degrees, speed=200):
    """หมุนมอเตอร์ตามองศาที่กำหนด"""
    # คำนวณจำนวน steps ที่ต้องหมุน
    total_steps = int((degrees / 360.0) * self.steps_per_revolution * self.microsteps)
    
    # กำหนดทิศทางการหมุน
    if degrees > 0:
        GPIO.output(self.dir_pin, GPIO.HIGH)  # หมุนตามเข็มนาฬิกา
        direction = 1
    else:
        GPIO.output(self.dir_pin, GPIO.LOW)   # หมุนทวนเข็มนาฬิกา
        direction = -1
    
    # คำนวณความล่าช้าระหว่าง steps
    delay = 1.0 / speed  # speed = steps per second
    
    # หมุนมอเตอร์
    for step in range(abs(total_steps)):
        GPIO.output(self.step_pin, GPIO.HIGH)
        time.sleep(delay / 2)
        GPIO.output(self.step_pin, GPIO.LOW)
        time.sleep(delay / 2)
    
    # อัปเดตตำแหน่งปัจจุบัน
    self.current_position += degrees
    self.current_position = self.current_position % 360  # จำกัดอยู่ที่ 0-360 องศา
    
    print(f"Motor moved {degrees} degrees. Current position: {self.current_position:.1f}°")

การทำงาน:
- คำนวณจำนวน steps ที่ต้องหมุนจากองศาที่กำหนด
- กำหนดทิศทางการหมุนตามค่าบวกหรือลบ
- สร้าง pulse signal สำหรับ step pin
- อัปเดตตำแหน่งปัจจุบันของมอเตอร์

def return_to_home(self, speed=200):
    """หมุนกลับตำแหน่งเริ่มต้น (0 องศา)"""
    if self.current_position != 0:
        # คำนวณองศาที่ต้องหมุนกลับ
        degrees_to_home = -self.current_position
        
        # หมุนกลับ
        self.move_degrees(degrees_to_home, speed)
        
        print(f"Motor returned to home position (0°)")
    else:
        print("Motor already at home position")
